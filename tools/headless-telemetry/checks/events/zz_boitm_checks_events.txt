# Headless telemetry (checks module, Test-Headless.ps1 -Checks), 2026-11-09.
# One-off checks of the plan 6 phase 6 fixes and D1, forcing each mechanism
# once and logging BOIFIX|<date>|<check>|<pass/fail/warn/info>|<detail>.
#   .1  C12: what ordered_scope_state does with save_scope_as and no position
#   .2/.3  D1: eligibility counts; force the event for 3 countries; a direct
#          multiplier test of the legitimacy modifier on Russia

# ── C12 ─────────────────────────────────────────────────────────
boitm_chk.1 = {
	type = country_event
	hidden = yes
	trigger = {
		c:FRA ?= this
		game_date >= 1836.2.1
		NOT = { has_global_variable = boitm_chk_1 }
	}
	immediate = {
		set_global_variable = boitm_chk_1
		set_variable = { name = boitm_chk_iter value = 0 }
		ordered_scope_state = {
			order_by = state_population
			root = { change_variable = { name = boitm_chk_iter add = 1 } }
			save_scope_as = chk_nopos
		}
		ordered_scope_state = {
			order_by = state_population
			position = 0
			check_range_bounds = no
			save_scope_as = chk_top
		}
		debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|c12_ordered_scope|info|iterations=[THIS.ScriptValue('boitm_chk_iterations')|0];states=[THIS.ScriptValue('boitm_states')|0]"
		# SCOPE.sState prints NULL_STATE in debug_log (run 1), so log from inside each state.
		scope:chk_nopos = { debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|c12_saved_nopos|info|state=[THIS.GetState.GetStateRegion.GetName];pop=[THIS.ScriptValue('boitm_chk_state_pop')|0]" }
		scope:chk_top = { debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|c12_saved_top|info|state=[THIS.GetState.GetStateRegion.GetName];pop=[THIS.ScriptValue('boitm_chk_state_pop')|0]" }
	}
}

# ── D1 ──────────────────────────────────────────────────────────
boitm_chk.2 = {
	type = country_event
	hidden = yes
	trigger = {
		game_date >= 1837.1.1
		NOT = { has_global_variable = boitm_chk_2 }
	}
	immediate = {
		set_global_variable = boitm_chk_2
		debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|sp_eligibility|info|spare30_with_works=[THIS.ScriptValue('boitm_chk_sp_spare30')|0];eligible=[THIS.ScriptValue('boitm_chk_sp_eligible')|0];town=[THIS.ScriptValue('boitm_chk_sp_town')|0];country=[THIS.ScriptValue('boitm_chk_sp_country')|0]"
		ordered_country = {
			limit = {
				NOT = { has_variable = boi_surplus_cooldown }
				any_scope_state = { boi_surplus_state_eligible = yes }
			}
			order_by = gdp
			max = 3
			check_range_bounds = no
			set_variable = boitm_chk_sp_forced
			if = {
				limit = { any_scope_state = { boi_surplus_state_eligible = yes boi_surplus_town_fits = yes } }
				trigger_event = { id = boi_surplus.10 }
			}
			else = {
				trigger_event = { id = boi_surplus.11 }
			}
		}
		# Run 1 used Russia, at the 100 cap. The lowest-legitimacy real country instead.
		ordered_country = {
			limit = {
				NOT = { is_country_type = decentralized }
				legitimacy < 60
				legitimacy > 10
			}
			order_by = boitm_chk_neg_legitimacy
			position = 0
			check_range_bounds = no
			set_variable = boitm_chk_legit_target
			debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|sp_multiplier_before|info|tag=[THIS.GetCountry.GetTagName];press=[THIS.ScriptValue('boitm_chk_press')|0];legitimacy=[THIS.ScriptValue('boitm_chk_legitimacy')|1]"
			add_modifier = {
				name = boi_surplus_government_works
				years = 5
				multiplier = boi_surplus_press_control
			}
		}
	}
}

boitm_chk.3 = {
	type = country_event
	hidden = yes
	trigger = {
		game_date >= 1837.3.1
		has_global_variable = boitm_chk_2
		NOT = { has_global_variable = boitm_chk_3 }
	}
	immediate = {
		set_global_variable = boitm_chk_3
		every_country = {
			limit = { has_variable = boitm_chk_sp_forced }
			if = {
				limit = {
					has_variable = boi_surplus_cooldown
					OR = {
						boitm_chk_sp_state_mods >= 1
						boitm_chk_sp_gov >= 1
					}
				}
				debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|sp_forced_event|pass|tag=[THIS.GetCountry.GetTagName];state_mods=[THIS.ScriptValue('boitm_chk_sp_state_mods')|0];gov=[THIS.ScriptValue('boitm_chk_sp_gov')|0];cooldown=[THIS.ScriptValue('boitm_chk_sp_cooldown')|0]"
			}
			else = {
				debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|sp_forced_event|fail|tag=[THIS.GetCountry.GetTagName];state_mods=[THIS.ScriptValue('boitm_chk_sp_state_mods')|0];gov=[THIS.ScriptValue('boitm_chk_sp_gov')|0];cooldown=[THIS.ScriptValue('boitm_chk_sp_cooldown')|0]"
			}
		}
		every_country = {
			limit = { has_variable = boitm_chk_legit_target }
			debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|sp_multiplier_after|info|tag=[THIS.GetCountry.GetTagName];press=[THIS.ScriptValue('boitm_chk_press')|0];legitimacy=[THIS.ScriptValue('boitm_chk_legitimacy')|1];gov=[THIS.ScriptValue('boitm_chk_sp_gov')|0]"
		}
	}
}

boitm_chk.7 = {
	type = country_event
	hidden = yes
	trigger = {
		game_date >= 1837.3.1
		NOT = { has_global_variable = boitm_chk_7 }
	}
	immediate = {
		set_global_variable = boitm_chk_7
		ordered_country = {
			limit = {
				NOT = { has_variable = boi_hev_great_fire_cd }
				NOT = { c:GBR ?= this }
				any_scope_state = {
					boi_hev_state_wood_fire = yes
					boitm_chk_industry_levels >= 1
				}
			}
			order_by = gdp
			position = 0
			check_range_bounds = no
			save_scope_as = chk_fire_country
		}
		if = {
			limit = { exists = scope:chk_fire_country }
			scope:chk_fire_country = {
				ordered_scope_state = {
					limit = { boi_hev_state_wood_fire = yes }
					order_by = boi_cm_state_low_tier_levels
					position = 0
					check_range_bounds = no
					save_scope_as = chk_fire_state
				}
				set_global_variable = { name = boitm_chk_fire_state value = scope:chk_fire_state }
				scope:chk_fire_state = {
					set_variable = { name = boitm_chk_ind_before value = boitm_chk_industry_levels }
					debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|fire_before|info|tag=[THIS.GetState.GetOwner.GetTagName];state=[THIS.GetState.GetStateRegion.GetName];low_tier=[THIS.ScriptValue('boi_cm_state_low_tier_levels')|0];wood=[THIS.ScriptValue('boitm_chk_wood_levels')|0];industry=[THIS.ScriptValue('boitm_chk_industry_levels')|0]"
				}
				trigger_event = { id = boi_housing_events.24 }
			}
		}
	}
}
