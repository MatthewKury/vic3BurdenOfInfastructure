# Headless telemetry (checks module), 2026-11-10.

# State scope. One line per peasant, laborer and farmer pop: its workforce and
# dependents. Farmers are the control for the per-pop-type keys: Statute
# Labour's keys name only peasants and laborers.
boitm_chk_log_dependents = {
	debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|dep_state|info|law=$LAW$;tag=[THIS.GetState.GetOwner.GetTagName];state=[THIS.GetState.GetStateRegion.GetName]"
	every_scope_pop = {
		limit = { is_pop_type = peasants }
		debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|dep_pop|info|law=$LAW$;type=peasants;workforce=[THIS.GetPop.GetNumWorkforce];dependents=[THIS.GetPop.GetDependentsSize]"
	}
	every_scope_pop = {
		limit = { is_pop_type = laborers }
		debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|dep_pop|info|law=$LAW$;type=laborers;workforce=[THIS.GetPop.GetNumWorkforce];dependents=[THIS.GetPop.GetDependentsSize]"
	}
	every_scope_pop = {
		limit = { is_pop_type = farmers }
		debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|dep_pop|info|law=$LAW$;type=farmers;workforce=[THIS.GetPop.GetNumWorkforce];dependents=[THIS.GetPop.GetDependentsSize]"
	}
}
