# 1 cheap, 2 dear, 0 neither, -1 no building.
boitm_st_choice_pm = {
	value = -1
	every_scope_building = {
		limit = { is_building_type = building_boitm_st_choice }
		value = 0
		if = { limit = { has_active_production_method = pm_boitm_st_cheap } value = 1 }
		else_if = { limit = { has_active_production_method = pm_boitm_st_dear } value = 2 }
	}
}

boitm_st_absent_short = {
	value = -1
	every_scope_building = {
		limit = { is_building_type = building_boitm_st_absent }
		value = 0
		if = { limit = { building_has_goods_shortage = yes } value = 1 }
	}
}
# Occupancy of the probe buildings in the state, lowest band (0.1 steps); -1 none.
boitm_st_occ = {
	value = -1
	every_scope_building = {
		limit = {
			OR = {
				is_building_type = building_boitm_st_choice
				is_building_type = building_boitm_st_choice_mp
				is_building_type = building_boitm_st_choice_same
				is_building_type = building_boitm_st_absent
				is_building_type = building_boitm_st_levels
			}
		}
		value = 0
		if = { limit = { occupancy >= 0.99 } value = 1 }
		else_if = { limit = { occupancy >= 0.9 } value = 0.9 }
		else_if = { limit = { occupancy >= 0.7 } value = 0.7 }
		else_if = { limit = { occupancy >= 0.5 } value = 0.5 }
		else_if = { limit = { occupancy >= 0.3 } value = 0.3 }
		else_if = { limit = { occupancy >= 0.1 } value = 0.1 }
	}
}
