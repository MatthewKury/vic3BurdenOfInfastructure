# Chargeable use (above the population floor), summed over states.
boitm_pw_chargeable = {
	value = 0
	every_scope_state = { add = boi_pw_chargeable }
}

# Population-weighted upkeep share x 100 (100 = penalty fully cancelled).
boitm_pw_share_pct = {
	value = 0
	every_scope_state = { add = { value = boi_pw_upkeep_share multiply = state_population } }
	divide = { value = 0 every_scope_state = { add = state_population } min = 1 }
	multiply = 100
}

# States whose hub cancels less than 90% of the penalty.
boitm_pw_states_under = {
	value = 0
	every_scope_state = { limit = { boi_pw_upkeep_share < 0.9 } add = 1 }
}
