# Headless telemetry (housing module): needs the boi_housing goods and the
# housing buildings. Housing goods are local, so price is per state; these
# count states in BOI's usual price bands. Country scope.

boitm_slum_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_boi_slums }
		add = level
	}
}

boitm_housing_expensive_states = {
	value = 0
	every_scope_state = {
		limit = { sg:boi_housing = { state_goods_pricier >= 0.5 } }
		add = 1
	}
}

boitm_housing_cheap_states = {
	value = 0
	every_scope_state = {
		limit = { sg:boi_housing = { state_goods_cheaper >= 0.25 } }
		add = 1
	}
}

# State scope: per-state housing log (boitm_housing.2).
boitm_hs_population = { value = state_population }
boitm_hs_infrastructure = { value = infrastructure }
boitm_hs_usage = { value = infrastructure_usage }
boitm_hs_urbanization = { value = total_urbanization }
boitm_hs_incorporated = {
	value = 0
	if = {
		limit = { is_incorporated = yes }
		value = 1
	}
}
boitm_hs_urban_center_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_urban_center }
		add = level
	}
}
boitm_hs_industry_levels = {
	value = 0
	every_scope_building = {
		limit = {
			OR = {
				is_building_group = bg_light_industry
				is_building_group = bg_heavy_industry
			}
		}
		add = level
	}
}
boitm_hs_residential_levels = {
	value = 0
	every_scope_building = {
		limit = {
			OR = {
				is_building_type = building_boi_slums
				is_building_type = building_boi_tenements
				is_building_type = building_boi_apartments
				is_building_type = building_boi_luxury_apartments
				is_building_type = building_boi_suburbs
				is_building_type = building_boi_temporary_shelter
			}
		}
		add = level
	}
}
