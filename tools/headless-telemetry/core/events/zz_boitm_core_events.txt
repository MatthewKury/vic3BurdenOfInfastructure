namespace = boitm

# Headless telemetry (core module). One debug_log line per metric group, so a
# bad data function blanks only its own line. Line format, parsed by
# tools/Get-HeadlessReport.ps1:
#   BOITM|<group>|<date>|<tag>|key=value;key=value
# Goods lines read the country's market: subjects in a customs union repeat
# their overlord's figures.
