namespace = boitm_fixture

# Headless telemetry (fixtures module): runs BOI's console-only housing debug
# events (boi_housing.2 to .4) as Portugal and logs one result line per check:
#   BOIFIX|<date>|<check>|<pass/fail/warn/info>|<detail>
# Kept off the BOITM| prefix so the yearly report ignores it; Test-Headless.ps1
# reads these lines and fails the run on any "fail". The checks follow
# docs/HOUSING_TESTING.md. Timeline from the first monthly pulse on or after
# 1836.2.1: Alentejo fixture (day 0), re-run (day 30), permissive laws (day 60),
# PM setup (day 61), protective laws (day 62), final check (day 97).

# Day 0: record Alentejo, then build the fixture.
boitm_fixture.1 = {
	type = country_event
	hidden = yes

	trigger = {
		c:POR ?= this
		game_date >= 1836.2.1
		NOT = { has_global_variable = boitm_fx_started }
		any_scope_state = { state_region = s:STATE_ALENTEJO }
	}

	immediate = {
		set_global_variable = boitm_fx_started
		random_scope_state = {
			limit = { state_region = s:STATE_ALENTEJO }
			set_variable = { name = boitm_fx_usage_before value = boitm_fx_usage }
			debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|start|info|alentejo_usage=[THIS.ScriptValue('boitm_fx_usage')|1]"
		}
		trigger_event = { id = boi_housing.2 days = 0 }
		trigger_event = { id = boitm_fixture.2 days = 30 }
	}
}

# Day 30: the five buildings exist and use 32 infrastructure; run it again.
boitm_fixture.2 = {
	type = country_event
	hidden = yes

	immediate = {
		random_scope_state = {
			limit = { state_region = s:STATE_ALENTEJO }
			if = {
				limit = {
					has_building = building_boi_slums
					has_building = building_boi_tenements
					has_building = building_boi_luxury_apartments
					has_building = building_boi_suburbs
					has_building = building_boi_temporary_shelter
				}
				debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|fixture_buildings|pass|levels=[THIS.ScriptValue('boitm_fx_housing_levels')|0]"
			}
			else = {
				debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|fixture_buildings|fail|levels=[THIS.ScriptValue('boitm_fx_housing_levels')|0]"
			}
			# Warn, not fail: other construction in Alentejo also moves usage.
			if = {
				limit = {
					boitm_fx_usage_delta >= 14.4
					boitm_fx_usage_delta <= 14.6
				}
				debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|fixture_infrastructure_usage|pass|delta=[THIS.ScriptValue('boitm_fx_usage_delta')|1]"
			}
			else = {
				debug_log = "BOIFIX|[TimeKeeper.GetCurrentDate.GetString]|fixture_infrastructure_usage|warn|delta=[THIS.ScriptValue('boitm_fx_usage_delta')|1]"
			}
			set_variable = { name = boitm_fx_levels_after_first value = boitm_fx_housing_levels }
		}
		trigger_event = { id = boi_housing.2 days = 0 }
		trigger_event = { id = boitm_fixture.3 days = 30 }
	}
}
