namespace = boitm_probe

# Engine probes for the 3 Nov 2026 work, run as France from 1836.2.1. Logs
# BOIPROBE|<date>|<probe>|<values>. Read them with grep in the run's debug.log.
#
#  Brittany  free      no input: the control for staffing-scaled infrastructure
#  Aquitaine absent    input good absent from the world (electricity)
#  Languedoc short     input good heavily over-demanded (transportation)
#  Burgundy  gate/tech start_building_construction against possible and tech gates
#            req       activate_production_method against required_input_goods
#  Lorraine  levels    add_building_level -1, then remove_building, on a 3-level building
#  Alsace-L. gate      queue a further level on a building already failing possible
#  Ile-de-F. tenement  add_building_level -1 on a seeded, country-owned housing building

boitm_probe.1 = {
	type = country_event
	hidden = yes
	trigger = {
		c:FRA ?= this
		game_date >= 1836.2.1
		NOT = { has_global_variable = boitm_probe_started }
	}
	immediate = {
		set_global_variable = boitm_probe_started
		random_scope_state = { limit = { state_region = s:STATE_BRITTANY } create_building = { building = building_boitm_probe_free level = 1 } }
		random_scope_state = { limit = { state_region = s:STATE_AQUITAINE } create_building = { building = building_boitm_probe_absent level = 1 } }
		random_scope_state = { limit = { state_region = s:STATE_LANGUEDOC } create_building = { building = building_boitm_probe_short level = 1 } }
		random_scope_state = {
			limit = { state_region = s:STATE_BURGUNDY }
			start_building_construction = building_boitm_probe_gate
			start_building_construction = building_boitm_probe_tech
			create_building = { building = building_boitm_probe_req level = 1 }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_LORRAINE }
			create_building = { building = building_boitm_probe_levels level = 3 }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_ALSACE_LORRAINE }
			create_building = { building = building_boitm_probe_gate level = 1 }
		}
		trigger_event = { id = boitm_probe.2 days = 2 }
		trigger_event = { id = boitm_probe.3 days = 30 }
		trigger_event = { id = boitm_probe.4 days = 60 }
		trigger_event = { id = boitm_probe.5 days = 62 }
		trigger_event = { id = boitm_probe.6 days = 90 }
		trigger_event = { id = boitm_probe.7 days = 92 }
		trigger_event = { id = boitm_probe.8 days = 100 }
		trigger_event = { id = boitm_probe.9 days = 104 }
	}
}
