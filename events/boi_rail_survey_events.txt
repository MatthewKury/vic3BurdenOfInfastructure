# ================================================================
#  Burden of Infrastructure — Rail Network Survey events
#
#  Design: docs/plans/01_RAIL_SURVEY_PROPOSAL.md. All numbers are
#  test values.
#
#   .1        choose a continent (from the decision)
#   .2 - .7   choose a strategic region on that continent
#   .9        hidden: start the survey
#   .11 - .14 first stage (25%): routes
#   .21 - .24 second stage (50%): land
#   .31 - .34 third stage (75%): approval
#   .40       Militarized Police only, second or third stage
#   .50       hidden monthly: keep the Rail Works cap modifiers right
#   .60       hidden yearly: the AI's stand-in for surveys
#   .90       completion: award the levels
#
#  Stage events act on scope:survey_state (a random surveyed state) or
#  on every surveyed state, never elsewhere; only interest group
#  approval is country-wide. "Force them out" options scale with the
#  police law and institution (boi_rs_forced_* values).
# ================================================================

boi_rail_survey.11 = {
	type = country_event
	placement = scope:survey_state
	title = boi_rail_survey.11.t
	desc = boi_rail_survey.11.d
	flavor = boi_rail_survey.11.f
	event_image = { video = "unspecific_trains" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = { boi_rs_has_surveyed_state = yes }

	immediate = {
		random_scope_state = {
			limit = { has_variable = boi_in_rail_survey }
			save_scope_as = survey_state
		}
	}

	# The long route: dear, but a better line.
	option = {
		name = boi_rail_survey.11.a
		default_option = yes
		add_treasury = boi_rs_payment_medium
		boi_rs_quality_add = { AMOUNT = 10 TT = up_10 }
		ai_chance = { base = 2 }
	}

	# The steep gradient: quick and cheap, a worse line.
	option = {
		name = boi_rail_survey.11.b
		boi_rs_quality_add = { AMOUNT = -10 TT = down_10 }
		boi_rs_progress_add = { AMOUNT = 5 TT = up_5 }
		ai_chance = { base = 1 }
	}
}

# Yearly, AI only: the stand-in for a survey. One year in four, the up to
# five states most short of infrastructure with Rail Works at the cap gain
# 3 levels (a mid-quality result), for a year of survey expenses paid at
# once. Skipped when the treasury can't cover it.
boi_rail_survey.60 = {
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		in_default = no
		has_technology_researched = railways
		NOT = { has_journal_entry = je_boi_rail_survey }
		any_scope_state = { boi_rs_ai_state_wants_cap = yes }
	}

	immediate = {
		random_list = {
			3 = {}
			1 = {
				set_variable = { name = boi_rs_ai_state_count value = 0 }
				ordered_scope_state = {
					limit = { boi_rs_ai_state_wants_cap = yes }
					order_by = boi_ai_infrastructure_shortfall
					max = 5
					check_range_bounds = no
					set_variable = boi_rs_ai_pick
					root = { change_variable = { name = boi_rs_ai_state_count add = 1 } }
				}
				# A debt test, not gold: AI countries that borrow hold almost no
				# gold, so gold_reserves > cost skipped nearly all of them. In the
				# 2026-11-08 smoke run France had £0 in gold at 3% scaled debt.
				# The charge goes on credit when there is no gold.
				if = {
					limit = { scaled_debt < 0.5 }
					add_treasury = {
						value = boi_rs_ai_cost
						multiply = -1
					}
					every_scope_state = {
						limit = { has_variable = boi_rs_ai_pick }
						if = {
							limit = { has_variable = boi_rail_cap_surveyed }
							change_variable = { name = boi_rail_cap_surveyed add = 3 }
						}
						else = {
							set_variable = { name = boi_rail_cap_surveyed value = 3 }
						}
						boi_rail_cap_refresh = yes
					}
					if = {
						limit = { has_variable = boi_rs_ai_surveys }
						change_variable = { name = boi_rs_ai_surveys add = 1 }
					}
					else = {
						set_variable = { name = boi_rs_ai_surveys value = 1 }
					}
				}
				every_scope_state = {
					limit = { has_variable = boi_rs_ai_pick }
					remove_variable = boi_rs_ai_pick
				}
				remove_variable = boi_rs_ai_state_count
			}
		}
	}
}
