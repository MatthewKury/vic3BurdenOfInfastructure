# Explicit debug fixture only. Run while playing Portugal; does not seed the world.
# Repeated invocations never add further levels to an existing fixture building.
boi_housing.2 = {
 type = country_event
 hidden = yes
 immediate = {
  every_scope_state = {
   limit = { state_region = s:STATE_ALENTEJO }
   if = {
    limit = { NOT = { has_building = building_boi_slums } }
    create_building = { building = building_boi_slums level = 1 }
   }
   if = {
    limit = { NOT = { has_building = building_boi_tenements } }
    create_building = { building = building_boi_tenements level = 1 }
   }
   if = {
    limit = { NOT = { has_building = building_boi_luxury_apartments } }
    create_building = { building = building_boi_luxury_apartments level = 1 }
   }
   if = {
    limit = { NOT = { has_building = building_boi_suburbs } }
    create_building = { building = building_boi_suburbs level = 1 }
   }
   if = {
    limit = { NOT = { has_building = building_boi_temporary_shelter } }
    create_building = { building = building_boi_temporary_shelter level = 1 }
   }
  }
 }
}

# Monthly: company housing follows the Labour Rights law (2026-11-08). The trigger
# is true only when the law differs from the level last enforced, so the
# conversion runs once per law change.
boi_housing.10 = {
 type = country_event
 hidden = yes
 trigger = {
  NOT = { is_country_type = decentralized }
  OR = {
   NOT = { has_variable = boi_labour_rights_level }
   AND = {
    has_law_or_variant = law_type:law_worker_protections
    has_variable = boi_labour_rights_level
    NOT = { var:boi_labour_rights_level = 3 }
   }
   AND = {
    has_law_or_variant = law_type:law_regulatory_bodies
    has_variable = boi_labour_rights_level
    NOT = { var:boi_labour_rights_level = 2 }
   }
   AND = {
    NOT = { has_law_or_variant = law_type:law_worker_protections }
    NOT = { has_law_or_variant = law_type:law_regulatory_bodies }
    has_variable = boi_labour_rights_level
    NOT = { var:boi_labour_rights_level = 1 }
   }
  }
 }
 immediate = { boi_housing_labour_rights_sync_effect = yes }
}
