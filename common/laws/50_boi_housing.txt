law_boi_mandated_serviced_housing = {
 group = lawgroup_boi_employer_housing
 icon = "gfx/interface/icons/law_icons/worker_protections.dds"
 progressiveness = 50
 unlocking_technologies = { socialism }
 # Serviced Town needs modern_sewerage, which socialism does not require.
 # Without it the effect moves paying tiers to No Housing instead.
 on_activate = { boi_housing_enforce_serviced_towns_effect = yes }
 # A person may still enact it without the tech; the AI waits so it doesn't end
 # all its company housing by accident.
 ai_will_do = { has_technology_researched = modern_sewerage }
 ai_enact_weight_modifier = { #Petitions
  value = 0
  if = {
   limit = {
    OR = {
     ig:ig_trade_unions ?= { is_in_government = yes }
     ig:ig_intelligentsia ?= { is_in_government = yes }
    }
   }
   add = 150
  }
  if = {
   limit = { ai_has_enact_weight_modifier_journal_entries = yes }
   add = 750
  }
 }
}

law_boi_residential_standards = {
 group = lawgroup_boi_residential_standards
 icon = "gfx/interface/icons/law_icons/regulatory_bodies.dds"
 progressiveness = 25
 # Blocks construction through the building's possible trigger.
 # Does not disable the fixed slum PM or remove existing buildings.
 # The law bans the cheap working-class types, so the AI waits for Regulated Terraces,
 # or it would leave itself with nothing cheap to build.
 ai_will_do = { has_technology_researched = urban_planning }
 ai_enact_weight_modifier = { #Petitions
  value = 0
  if = {
   limit = {
    OR = {
     ig:ig_trade_unions ?= { is_in_government = yes }
     ig:ig_intelligentsia ?= { is_in_government = yes }
     ig:ig_devout ?= { is_in_government = yes }
     ig:ig_armed_forces ?= { is_in_government = yes }
    }
   }
   add = 150
  }
  # Sanitary science makes the case, and Model Dwellings become an alternative.
  if = {
   limit = { has_technology_researched = modern_sewerage }
   add = 150
  }
  if = {
   limit = { ai_has_enact_weight_modifier_journal_entries = yes }
   add = 750
  }
 }
}
