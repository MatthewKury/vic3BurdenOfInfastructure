# ================================================================
#  Burden of Infrastructure - Water goods
#
#  Design: docs/plans/03_WATER.md. Two goods, one price.
#
#  water       LOCAL. What households (and later industry) consume.
#              Cannot leave the state it is made in, so a state's
#              supply and demand have to meet inside its borders.
#
#  bulk_water  NOT local, tradeable = no. Water in transit: the only
#              way Local Water crosses a state border is by being
#              converted to Bulk Water on one side (export PMs) and
#              back to Local Water on the other (import PMs).
#
#  Both cost 20, as agreed: the conversions are lossy, so a round
#  trip always loses water and nobody can print money by looping
#  one state's supply through the market.
#
#  TEST TARGET: tradeable = no is used by vanilla only for gold,
#  which is also fixed_price. Whether it keeps a good circulating
#  between the states of ONE market while blocking world-market
#  trade is UNVERIFIED. If bulk_water turns out to be unsellable
#  even between neighbours, remove the flag - see tools/water-
#  prototype/README.md for the checklist.
#
#  The goods_input_/goods_output_ modifier types are declared by
#  hand in common/modifier_type_definitions/51_boi_water_modifier_types.txt;
#  without them every production method referencing the goods is
#  rejected at load (lessons file, section 13).
# ================================================================

water = {
	texture = "gfx/interface/icons/goods_icons/boi_water.dds"
	cost = 20
	category = staple
	local = yes

	consumption_tax_cost = 200
}

bulk_water = {
	texture = "gfx/interface/icons/goods_icons/boi_bulk_water.dds"
	cost = 20
	category = staple
	tradeable = no
}
