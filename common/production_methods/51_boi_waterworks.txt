# ================================================================
#  Burden of Infrastructure - Waterworks production methods
#
#  Magnitudes are PLACEHOLDERS to calibrate in play. The ratios,
#  signs and gates are the design. Prices: water 20, bulk_water 20,
#  horses 30, coal 30, tools 40, iron 40, electricity 30.
#
#  Structure (see docs/plans/03_WATER.md):
#
#   SOURCE  produce here, or import. Per-level employment is
#           level_scaled; goods and state modifiers are
#           workforce_scaled, so an understaffed works produces and
#           consumes proportionally less (lessons file, s.13).
#
#   EXPORT  converts Local Water to Bulk Water at 75% (25% lost to
#           leakage and evaporation). NO building_employment_* key
#           appears in any export PM, so choosing an export level
#           never changes staffing. It draws on the state's whole
#           water pool at the state price, not just this building's
#           output, so a short state sees the price rise and export
#           stop paying - scarcity shows itself.
#
#   ADMIN   Bureaucracy law, same gates as Public Works.
#
#  LOSSES: import is ~75% efficient (bulk in : water out), export is
#  75%, so a round trip returns ~56% before carriage. With both
#  goods at 20 there is no arbitrage loop.
#
#  Period anchors (research in docs/archive/planning-2026-10-24/PROPOSAL_WATER.md):
#    - wooden mains and intermittent supply in early-19th-c. London
#    - slow sand filtration, Chelsea 1829
#    - cholera 1832 / 1849 / 1854; Broad Street pump traced to a
#      cesspit-contaminated well -> wells carry a mortality cost
#    - steam pumping and cast-iron mains, Liverpool 1847
#    - Metropolis Water Act 1852: upstream intakes, filtered supply
#    - routine disinfection of public supplies, c. 1908
#
#  STRESS OF INFRASTRUCTURE: every level also costs the group's
#  infrastructure_usage_per_level = 2 (building group). The piped
#  tiers add a further negative state_infrastructure_add because
#  mains and pumping stations occupy the same network the works
#  are trying to extend.
# ================================================================

# Tier 1: Wells and Water Carriers (default, 1836, no tech gate).
# Shallow wells, standpipes and water-sellers' carts. Unreliable and
# unsafe: the Broad Street pump drew from one of these.
# Horses haul the barrels. The mortality entry is the Broad Street
# cost of drinking from the nearest well.

pm_boi_wells_and_carriers = {
	texture = "gfx/interface/icons/production_method_icons/anchorage.dds"

	building_modifiers = {
		workforce_scaled = {
			goods_input_horses_add = 1       # 30
			goods_output_water_add = 24      # 480
		}

		level_scaled = {
			building_employment_laborers_add = 700
		}
	}

	state_modifiers = {
		workforce_scaled = {
			state_mortality_mult = 0.005
		}
	}
}

# ================================================================
#  PART 1b - SOURCE: import (Bulk Water in, Local Water out)
#
#  About 75% efficient before carriage, so producing locally is
#  cheaper wherever the state has water. Importing is a remedy for
#  scarcity, not a free alternative. Staffing is lower than local
#  production: a pipe needs fewer hands than a pumping station.
# ================================================================
