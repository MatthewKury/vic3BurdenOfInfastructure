# ================================================================
#  Burden of Infrastructure — Road Works PMs
#
#  building_boi_infrastructure is Road Works (the key predates the
#  split from Rail Works; the 1836 seed and its audits use it).
#
#  THREE groups, deliberately independent:
#
#   pmg_boi_infrastructure_method  (below, part 1)
#       400 technical staff, goods in, infrastructure out.
#       Follows TECHNOLOGY.
#
#   pmg_boi_road_administration    (below, part 2)
#       100 administrative staff. Follows the BUREAUCRACY LAW,
#       changing WHO holds the jobs. Same headcount either way.
#
#   pmg_boi_road_financing         (below, part 3)
#       No staff of its own. Follows the FINANCING LAW, changing
#       WHO PAYS: discounts and their side effects. Rail Works and
#       Harbour Works have their own groups on the same laws.
#
#  These are COST CENTRES: no goods_output_ line. Infrastructure is a
#  STATE MODIFIER, and a shortage of inputs cuts it (probe, 2026-11-03).
#
#  Prices for the "# N" comments: horses 30, tools 40, coal 30,
#  oil 40, automobiles 100.
# ================================================================


# ================================================================
#  PART 1 — Method ladder (400 technical staff, follows tech)
#
#  Laborers 400 -> 125, machinists 0 -> 175, engineers 0 -> 100.
#  Technical headcount stays flat at 400; only the skill mix moves.
#  Roads run on horses, then automobiles: no transportation at all.
#  No required_input_goods on automobiles or oil: it only works on
#  local goods (probe, 2026-11-03), so a shortage penalty does the job.
# ================================================================

# ================================================================
#  PART 2 — Administration (100 staff, follows Bureaucracy law)
#
#  Same 100 jobs in every case; only WHO holds them changes.
#  Aristocratic -> bureaucratic -> clerical.
#
#  This gives the Bureaucracy law a real political consequence:
#  under Hereditary Bureaucrats every state employs 50 aristocrats
#  per infrastructure level, which is a large state-funded boost to
#  Landowner political strength across a whole empire.
#
#  pm_boi_appointed_surveyors is the guaranteed fallback: it uses
#  disallowing_laws rather than an unlocking_laws gate, so it stays
#  valid whenever the other two do not apply. One PM in a group
#  must ALWAYS be valid.
# ================================================================


# ── Appointed Bureaucrats (default / fallback) ──────────────────
# County surveyors and a professional civil service.

pm_boi_appointed_surveyors = {
	texture = "gfx/interface/icons/production_method_icons/professional_bureaucrats.dds"

	is_default = yes

	disallowing_laws = {
		law_hereditary_bureaucrats
		law_crownland_diets
		law_elected_bureaucrats
	}

	building_modifiers = {
		level_scaled = {
			building_employment_bureaucrats_add = 50
			building_employment_clerks_add = 50
		}
	}
}

# ================================================================
#  PART 3 — Financing (no staff of its own, follows the financing law)
#
#  Who pays. Each PM is unlocked by exactly one law in
#  lawgroup_boi_road_financing (display name "Public Works Financing";
#  the keys predate the rename), and the state-funded PM is the
#  fallback via disallowing_laws. Road, Rail and Harbour Works each
#  have their own group, because one shared group would stack the
#  +15% and apply one flat cut to very different outputs.
#
#  Only effects that should GROW with the network live here. Flat
#  country-wide effects sit on the laws, so they cannot stack.
#  State funding changes nothing: it is the BALANCE.md baseline.
# ================================================================


pm_boi_public_highways = {
	texture = "gfx/interface/icons/production_method_icons/professional_bureaucrats.dds"

	is_default = yes

	disallowing_laws = {
		law_boi_statute_labour
		law_boi_turnpike_trusts
		law_boi_road_fund # BOI 2026-11-08: Road Works use pm_boi_road_fund instead
	}
}


# Dedicated Road Fund (2026-11-08): motor taxes pay for the roads, so the
# state buys 25% fewer goods. No other change; Rail and Harbour Works keep
# their state-funding PMs under this law.
pm_boi_road_fund = {
	texture = "gfx/interface/icons/production_method_icons/professional_bureaucrats.dds"

	unlocking_laws = {
		law_boi_road_fund
	}

	building_modifiers = {
		unscaled = {
			building_goods_input_mult = -0.25
		}
	}
}


# Tolls pay for upkeep, so the state buys 35% fewer goods, and the
# trust's gangs replace 60 state-paid labourers per level with 50
# toll-keepers and 10 investors. The +15% lives on Road Works only, so
# it stays once per state. UNSCALED, and it multiplies all of the
# state's infrastructure. The market-access cost is on the law.
pm_boi_turnpike_tolls = {
	texture = "gfx/interface/icons/production_method_icons/merchant_guilds.dds"

	unlocking_laws = {
		law_boi_turnpike_trusts
	}

	building_modifiers = {
		unscaled = {
			building_goods_input_mult = -0.35
			# Stands in for toll income: no modifier can route money
			# to a specific pop, and government buildings have no
			# revenue. Applies only to shopkeepers in this building.
			building_shopkeepers_standard_of_living_add = 2
		}

		level_scaled = {
			# Must stay below the smallest laborer count (125,
			# Modern Highways) or employment goes negative.
			building_employment_laborers_add = -60
			building_employment_shopkeepers_add = 50
			building_employment_capitalists_add = 10
		}
	}

	state_modifiers = {
		unscaled = {
			state_infrastructure_mult = 0.15
		}
	}
}


# Peasants owe road days and bring their own teams: 30% fewer goods
# bought and 2 less infrastructure per level. The law itself carries the
# human cost (fewer working adults among peasants and laborers) and the
# estate bonus, since the squire decides where the duty goes.
pm_boi_statute_labour = {
	texture = "gfx/interface/icons/production_method_icons/corvee_labor.dds"

	unlocking_laws = {
		law_boi_statute_labour
	}

	building_modifiers = {
		unscaled = {
			building_goods_input_mult = -0.3
		}
	}

	# The human cost and the estate bonus are on the law, country-wide
	# (2026-11-10: -0.02 working-adult ratio for peasants and laborers,
	# +5% farm and ranch throughput). Peasant mortality and the bonus were
	# here, per level (+3% mortality, +1% throughput) until 2026-11-09, then
	# a taper (review C8), because in p6_iter1 states with 8+ Road Works lost
	# population. The infrastructure cut stays per level: it is the output
	# reduction itself.
	state_modifiers = {
		workforce_scaled = {
			state_infrastructure_add = -2
		}
	}
}
