# demand. Each tier's inputs and output scale together, which keeps its break-even
# housing price (and so the profit order between tiers) unchanged.

 texture = "gfx/interface/icons/production_method_icons/wooden_buildings.dds"
 unlocking_laws = { law_no_workers_rights } # Labour Rights decides company housing (2026-11-08)
 building_modifiers = {
  workforce_scaled = {
   goods_input_wood_add = 0.075
   goods_input_fabric_add = 0.075
   goods_output_boi_housing_add = 0.45
  }
  # The flat +15% mortality on employed professions was removed 2026-11-10.
  # Camps put people off instead: -0.2% migration attraction per staffed
  # level (as Slums). A cold-scaled cost (1.15 Cold, cold_effect) was
  # considered and shelved the same day.
 }
 state_modifiers = {
  workforce_scaled = {
   state_migration_pull_mult = -0.002
  }
 }
}
