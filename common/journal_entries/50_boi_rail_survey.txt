# ================================================================
#  Burden of Infrastructure — Rail Network Survey journal entry
#
#  Added only by add_journal_entry from boi_rs_start, after the player
#  picks a region. Progress (0-100) and quality (0-100) are country
#  variables advanced in on_monthly_pulse (boi_rs_monthly); the bars
#  mirror them. Events fire at 25%, 50% and 75%. Ends when progress
#  reaches 100 (levels awarded by boi_rail_survey.90), when cancelled, or
#  when every surveyed state is lost.
#  Design: docs/plans/01_RAIL_SURVEY_PROPOSAL.md.
# ================================================================

je_boi_rail_survey = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_internal_affairs

	# Required by the engine even for a JE only ever added by effect.
	activation_description = {
		custom_tooltip = je_boi_rail_survey_activation_tt
	}

	scripted_progress_bar = boi_rs_progress_bar
	scripted_progress_bar = boi_rs_quality_bar

	scripted_button = boi_rs_cancel_button

	should_be_pinned_by_default_uninvolved_or_context = yes

	# Weekly expenses: £100 per surveyed state plus GDP / 50,000,
	# re-evaluated as GDP moves.
	scaling_modifiers_while_active = {
		modifier = boi_rail_survey_expenses
		value = boi_rs_weekly_cost
	}

	on_monthly_pulse = {
		effect = {
			boi_rs_monthly = yes
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_boi_rail_survey_completed_desc
		color = "gold_positive"
		trigger = {
			custom_tooltip = {
				text = boi_rs_complete_tt
				var:boi_rs_progress >= 100
			}
		}
		effect = {
			custom_tooltip = boi_rs_complete_effect_tt
			hidden_effect = {
				trigger_event = { id = boi_rail_survey.90 popup = yes }
			}
		}
	}

	outcome = {
		header = boi_rs_cancelled_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_boi_rail_survey_cancelled_desc
		color = "blue_negative"
		trigger = {
			custom_tooltip = {
				text = boi_rs_cancelled_tt
				has_variable = boi_rs_cancelled
			}
		}
		effect = {
			custom_tooltip = boi_rs_cancelled_effect_tt
			hidden_effect = {
				boi_rs_cleanup = yes
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_boi_rail_survey_lost_desc
		color = "blue_negative"
		trigger = {
			custom_tooltip = {
				text = boi_rs_lost_tt
				NOT = { boi_rs_has_surveyed_state = yes }
			}
		}
		effect = {
			custom_tooltip = boi_rs_lost_effect_tt
			hidden_effect = {
				boi_rs_cleanup = yes
			}
		}
	}

	weight = 100
	transferable = no
	can_revolution_inherit = no
}
