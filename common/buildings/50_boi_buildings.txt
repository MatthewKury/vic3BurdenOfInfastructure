# ================================================================
#  Burden of Infrastructure — Buildings
#
#  Four new buildings. The separate railway shadow adjusts vanilla AI.
#
#  building_boi_horse_ranch      bg_ranching, private, no tech gate
#  building_boi_infrastructure   Road Works, govt-funded, everywhere
#  building_boi_rail_works       Rail Works, govt-funded, needs a railway
#  building_boi_harbour_works    Harbour Works, govt-funded, coastal only
#
#  The three works buildings are pure COST CENTRES, built on
#  the same pattern as vanilla's Government Administration: they
#  emit a state modifier (state_infrastructure_add) instead of a
#  salable good, so they have no goods_output at all. The state
#  pays the wages and buys the inputs. See BALANCE.md.
#
#  The works buildings have no explicit maximum level, matching
#  vanilla Railways. Their inherited stateregion_max_level setting only changes
#  the scope of a cap when has_max_level = yes, which none of these buildings
#  sets. This is an intentional assumption to revisit after AI/balance testing.
#  Neither group declares infrastructure_usage_per_level, so an infrastructure
#  building does not itself consume infrastructure - which is correct.
# ================================================================


# ── Horse Ranch ─────────────────────────────────────────────────
# The horse supply. Deliberately mirrors building_livestock_ranch
# (same group, same city_type, same terrain, same construction
# cost) so the two compete on even terms for arable land.
#
# CRITICAL: no unlocking_technologies. Vanilla's livestock ranch
# requires 'enclosure', but horses are the 1836 infrastructure
# input - if this were tech-gated the game would open with no horse
# supply, no working Infrastructure buildings, and a global
# infrastructure collapse. It must be buildable on day one.
#
# bg_ranching supplies should_auto_expand = default_auto_expand_rule,
# so private capital will grow ranches on its own once demand exists.

building_boi_horse_ranch = {
	building_group = bg_ranching

	icon = "gfx/interface/icons/building_icons/boi_horse_ranch.dds"

	# Map model (2026-11-08): the culture's Livestock Ranch model. tools/Build-CityTypes.py lists this
	# building in every farm city type file; a building with a city_type must be
	# in every file of its hub type or the game asserts "Failed creating backing
	# building". Re-run the script after a game patch.
	city_type = farm
	levels_per_mesh = 50

	# pmg_boi_company_housing is the housing module's employer-housing
	# group, written here by hand. tools/Build-Housing.ps1 adds it to
	# vanilla rural buildings only; a generated copy of this block
	# would sort first (00_) and silently replace this definition.
	# The by-product and fencing groups match vanilla's Livestock Ranch
	# (2026-11-08); pmg_fencing is vanilla's own group. Its refrigeration
	# group is left out: horses travel live.
	production_method_groups = {
		pmg_boi_company_housing
		pmg_boi_horse_ranch_method
		pmg_boi_horse_ranch_byproduct
		pmg_fencing
	}

	required_construction = construction_cost_low

	terrain_manipulator = pasture
	ownership_type = self

	# Horses gate the entire infrastructure system, so the AI needs
	# to treat ranches as more than an ordinary farm. Below railway
	# (2000) but well above a default rural building. Scaled by the
	# local horse price the same way the railway shadow scales by
	# transportation: expensive horses raise priority (x1.1 / x1.2),
	# cheap horses lower it (x0.9 / x0.8).
	ai_value = {
		value = 1200
		multiply = boi_ai_horse_output_price
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_farming.dds"
}


# ── Road Works ──────────────────────────────────────────────────
# The heart of the mod. Converts horses, later automobiles, into state
# infrastructure at a permanent net loss. The key predates the split
# from Rail Works; the 1836 seed and its audits use it.
#
# Three production method groups on purpose:
#   pmg_boi_infrastructure_method  - 400 technical staff, goods in,
#                                    infrastructure out. Follows TECH.
#   pmg_boi_road_administration    - 100 admin staff. Follows the
#                                    BUREAUCRACY LAW, changing who
#                                    holds the jobs (aristocrats /
#                                    bureaucrats / clerks).
#   pmg_boi_road_financing         - no staff of its own. Follows the
#                                    FINANCING LAW: who pays
#                                    (peasants / tolls / treasury).
# Exactly one PM per group is active, which is what lets the three
# axes vary independently - the same structure vanilla uses on
# Government Administration.

building_boi_infrastructure = {
	# bg_boi_public_works, not bg_public_infrastructure directly -
	# no vanilla building occupies an intermediate group. See
	# common/building_groups/50_boi_building_groups.txt.
	building_group = bg_boi_public_works

	# Assumption: deliberately unlimited, matching building_railway. Revisit
	# after campaign testing establishes whether this permits overbuilding.

	# Stone bridge and tools distinguish Road Works from other
	# Development buildings and the coastal Harbour Works icon.
	icon = "gfx/interface/icons/building_icons/boi_public_works.dds"

	# Map model (2026-11-08): the culture's Construction Sector model, a builder's yard. tools/Build-CityTypes.py lists this
	# building in every city city type file; a building with a city_type must be
	# in every file of its hub type or the game asserts "Failed creating backing
	# building". Re-run the script after a game patch.
	city_type = city
	levels_per_mesh = 50

	production_method_groups = {
		pmg_boi_infrastructure_method
		pmg_boi_road_administration
		pmg_boi_road_financing
	}

	required_construction = construction_cost_low # was medium until 2026-11-08

	# The AI's budget cuts downsize government buildings regardless of
	# ai_value: headless runs saw Yorkshire cut from 2 levels to 0 at
	# -30 spare infrastructure, and whole countries drop every level in
	# one budget squeeze. Levels can only grow; see AI_VALIDATION.md.
	downsizeable = no

	# MUST be no_ownership: the engine rejects any building whose
	# group is is_government_funded = yes with a different
	# ownership_type ("Building type X is goverment funded but
	# ownership_type is not equal to no_ownership"). Government
	# buildings have no owner - the state simply runs them.
	ownership_type = no_ownership

	# State-scoped priority: infrastructure urgency dominates input prices.
	# INERT as far as tested: 2026-10-30 headless runs saw scores of
	# 14,400+ in deficit states with no construction. The AI construction
	# event (events/boi_ai_infrastructure_events.txt) does the real work.
	# Kept deliberately; reinvestigate whether the engine reads it at all
	# (docs/plans/01_CORE_INFRASTRUCTURE_AND_HORSES.md, step 3).
	ai_value = {
		value = 3000
		multiply = boi_ai_infrastructure_urgency
		multiply = boi_ai_public_works_input_price
	}

	# Auto-expansion needs a building with its toggle enabled. Grow only in
	# states with less than 5 spare infrastructure, after this building reaches
	# 90% staffing, never while this same building is already queued, and never
	# while the state owner is in default.
	should_auto_expand = {
		occupancy >= 0.9
		state = {
			infrastructure_delta < 5
			owner = { in_default = no }
			NOT = {
				any_scope_building = {
					is_building_type = building_boi_infrastructure
					is_under_construction = yes
				}
			}
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_railways.dds"
}


# ── Rail Works ──────────────────────────────────────────────────
# Turns a railway's transportation into state infrastructure. Needs a
# railway in the state: transportation is local, and railway states
# are the ones not short of it. possible blocks new and further levels,
# and logs a script error if queued anyway, so the AI event checks it
# first. start_building_construction ignores unlocking_technologies,
# so the event checks the tech too. Costs more to lay than a road, less
# than the railway itself (800).

building_boi_rail_works = {
	building_group = bg_boi_public_works

	icon = "gfx/interface/icons/building_icons/boi_rail_works.dds"

	# Map model (2026-11-08): the Army Logistics Centre's depot model; vanilla has no station model. tools/Build-CityTypes.py lists this
	# building in every city city type file; a building with a city_type must be
	# in every file of its hub type or the game asserts "Failed creating backing
	# building". Re-run the script after a game patch.
	city_type = city
	levels_per_mesh = 50

	unlocking_technologies = {
		railways
	}

	possible = {
		has_building = building_railway
	}

	production_method_groups = {
		pmg_boi_rail_works_method
		pmg_boi_rail_administration
		pmg_boi_rail_financing
	}

	required_construction = construction_cost_high

	# Capped per state (Rail Network Survey, docs/plans/01_RAIL_SURVEY_PROPOSAL.md).
	# The engine reads the cap from state_building_boi_rail_works_max_level_add,
	# declared in 50_boi_modifier_types.txt and set by the boi_rail_cap_*
	# static modifiers that events/boi_rail_survey_events.txt keeps on states.
	has_max_level = yes

	# Same reason as Road Works: AI budget cuts ignore ai_value.
	downsizeable = no

	ownership_type = no_ownership

	# Inert as far as tested; see the note on Road Works' ai_value.
	ai_value = {
		value = 2000
		multiply = boi_ai_infrastructure_urgency
		multiply = boi_ai_transportation_input_price
	}

	should_auto_expand = {
		occupancy >= 0.9
		state = {
			infrastructure_delta < 5
			owner = { in_default = no }
			has_building = building_railway
			boi_ai_rail_works_room > 0
			NOT = {
				any_scope_building = {
					is_building_type = building_boi_rail_works
					is_under_construction = yes
				}
			}
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_railways.dds"
}


# ── Harbour Works ───────────────────────────────────────────────
# Coastal-only second infrastructure avenue, paid for in merchant
# marine. This is the ONLY way a production method can be made
# conditional on geography: PMs have no trigger of any kind, so the
# condition has to live on the building via potential = { }.
#
# Deliberately government-funded rather than a PM group bolted onto
# vanilla's building_port: ports are privately owned, PM groups
# default to ai_selection = most_profitable, and a PM that produces
# a state modifier rather than a salable good has no profit signal -
# so AI-owned ports would never have selected it and the whole
# feature would have been player-only.

building_boi_harbour_works = {
	building_group = bg_boi_public_works

	# Assumption: deliberately unlimited, matching building_railway. Revisit
	# after campaign testing establishes whether this permits overbuilding.

	icon = "gfx/interface/icons/building_icons/boi_harbour_works.dds"

	# Map model (2026-11-08): the culture's lighthouse; vanilla's Port has no building model to copy. tools/Build-CityTypes.py lists this
	# building in every port city type file; a building with a city_type must be
	# in every file of its hub type or the game asserts "Failed creating backing
	# building". Re-run the script after a game patch.
	city_type = port
	levels_per_mesh = 50
	# The coastal restriction still comes from potential below.

	potential = {
		is_coastal = yes
	}

	production_method_groups = {
		pmg_boi_harbour_method
		pmg_boi_harbour_administration
		pmg_boi_harbour_financing
	}

	required_construction = construction_cost_low # was medium until 2026-11-08

	# Same reason as Road Works: AI budget cuts ignore ai_value.
	downsizeable = no

	# no_ownership, for the same reason as building_boi_infrastructure
	# above - required for every is_government_funded building group.
	ownership_type = no_ownership

	# Inert as far as tested; see the note on Road Works' ai_value.
	ai_value = {
		value = 1500
		multiply = boi_ai_infrastructure_urgency
		multiply = boi_ai_merchant_marine_input_price
	}

	# Same safeguard as Road Works, evaluated separately so either source can
	# be auto-expanded in a coastal state when its own toggle is enabled.
	should_auto_expand = {
		occupancy >= 0.9
		state = {
			infrastructure_delta < 5
			owner = { in_default = no }
			NOT = {
				any_scope_building = {
					is_building_type = building_boi_harbour_works
					is_under_construction = yes
				}
			}
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_harbour.dds"
}
