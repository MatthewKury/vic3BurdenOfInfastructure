# Company housing follows the vanilla Labour Rights law (2026-11-08); BOI's own
# Employer Housing law group is gone. No Workers' Rights permits Work Camps, Regulatory
# Bodies Work Towns, Worker Protections Serviced Towns (each plus No Housing).
# Vanilla's laws are not shadowed, so a monthly check (boi_housing.10) runs this and
# converts buildings when the law has changed. boi_labour_rights_level records the
# level last enforced: 1, 2 or 3.
boi_housing_labour_rights_sync_effect = {
 if = {
  limit = { has_law_or_variant = law_type:law_worker_protections }
  if = {
   limit = {
    OR = {
     NOT = { has_variable = boi_labour_rights_level }
     AND = { has_variable = boi_labour_rights_level NOT = { var:boi_labour_rights_level = 3 } }
    }
   }
   boi_housing_enforce_serviced_towns_effect = yes
   set_variable = { name = boi_labour_rights_level value = 3 }
  }
 }
 else_if = {
  limit = { has_law_or_variant = law_type:law_regulatory_bodies }
  if = {
   limit = {
    OR = {
     NOT = { has_variable = boi_labour_rights_level }
     AND = { has_variable = boi_labour_rights_level NOT = { var:boi_labour_rights_level = 2 } }
    }
   }
   boi_housing_enforce_work_towns_effect = yes
   set_variable = { name = boi_labour_rights_level value = 2 }
  }
 }
 else = {
  if = {
   limit = {
    OR = {
     NOT = { has_variable = boi_labour_rights_level }
     AND = { has_variable = boi_labour_rights_level NOT = { var:boi_labour_rights_level = 1 } }
    }
   }
   boi_housing_enforce_work_camps_effect = yes
   set_variable = { name = boi_labour_rights_level value = 1 }
  }
 }
}
