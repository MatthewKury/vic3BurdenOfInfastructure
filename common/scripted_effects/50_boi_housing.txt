# Country scope. Preserve existing housing laws on maintenance and load.
boi_housing_setup_effect = {
 ig:ig_trade_unions ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_reform } }
   add_ideology = ideology_boi_housing_reform
  }
 }
 ig:ig_intelligentsia ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_reform } }
   add_ideology = ideology_boi_housing_reform
  }
 }
 ig:ig_landowners ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_laissez_faire } }
   add_ideology = ideology_boi_housing_laissez_faire
  }
 }
 ig:ig_industrialists ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_laissez_faire } }
   add_ideology = ideology_boi_housing_laissez_faire
  }
 }
 ig:ig_devout ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_moral_reform } }
   add_ideology = ideology_boi_housing_moral_reform
  }
 }
 ig:ig_armed_forces ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_recruit_fitness } }
   add_ideology = ideology_boi_housing_recruit_fitness
  }
 }
 ig:ig_rural_folk ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_tied_cottages } }
   add_ideology = ideology_boi_housing_tied_cottages
  }
 }
 ig:ig_petty_bourgeoisie ?= {
  if = {
   limit = { NOT = { has_ideology = ideology:ideology_boi_housing_ratepayers } }
   add_ideology = ideology_boi_housing_ratepayers
  }
 }
 if = {
  limit = {
   NOR = {
    has_law = law_type:law_boi_unregulated_company_housing
    has_law = law_type:law_boi_protected_company_housing
    has_law = law_type:law_boi_mandated_serviced_housing
   }
  }
  activate_law = law_type:law_boi_unregulated_company_housing
 }
 if = {
  limit = {
   NOR = {
    has_law = law_type:law_boi_unregulated_residential
    has_law = law_type:law_boi_residential_standards
   }
  }
  activate_law = law_type:law_boi_unregulated_residential
 }
 if = {
  limit = { has_law = law_type:law_boi_unregulated_company_housing }
  boi_housing_enforce_work_camps_effect = yes
 }
 if = {
  limit = { has_law = law_type:law_boi_protected_company_housing }
  boi_housing_enforce_work_towns_effect = yes
 }
 if = {
  limit = { has_law = law_type:law_boi_mandated_serviced_housing }
  boi_housing_enforce_serviced_towns_effect = yes
 }
}
